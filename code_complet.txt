import { NextResponse } from 'next/server';
import { createClient } from '@/lib/supabase/server';
import { z } from 'zod';

const querySchema = z.object({
  username: z.string().min(3).regex(/^[a-zA-Z0-9_]+$/),
});

export async function GET(request: Request) {
  const { searchParams } = new URL(request.url);
  const username = searchParams.get('username');

  // Validation basique
  const result = querySchema.safeParse({ username });
  if (!result.success) {
    return NextResponse.json({ available: false, error: 'Format invalide' }, { status: 400 });
  }

  const supabase = await createClient();

  // On cherche si le pseudo existe dÃ©jÃ  (en ignorant la casse)
  const { data, error } = await supabase
    .from('users')
    .select('username')
    .ilike('username', username!) // ilike = insensible Ã  la casse
    .maybeSingle();

  if (error) {
    return NextResponse.json({ error: error.message }, { status: 500 });
  }

  // Si data existe, le pseudo est pris. Sinon, il est libre.
  return NextResponse.json({ available: !data });
}

export async function GET(request: Request) {
  const { searchParams, origin } = new URL(request.url);
  const code = searchParams.get("code");
  // "next" est la page oÃ¹ l'utilisateur voulait aller (ex: /dashboard)
  const next = searchParams.get("next");

  if (code) {
    const supabase = await createClient();
    const { error } = await supabase.auth.exchangeCodeForSession(code);

    if (!error) {
      // âœ… SÃ‰CURITÃ‰ : Protection contre les "Open Redirects"
      // On s'assure que 'next' commence par '/' mais pas par '//' (qui redirigerait vers un site externe)
      const isInternalUrl = next && next.startsWith("/") && !next.startsWith("//");
      
      // Si l'URL est sÃ»re, on y va. Sinon, direction le dashboard par dÃ©faut.
      const redirectUrl = isInternalUrl ? `${origin}${next}` : `${origin}/dashboard`;
      
      return NextResponse.redirect(redirectUrl);
    }
  }

  // Si erreur ou pas de code, retour Ã  la page de login avec une erreur
  return NextResponse.redirect(`${origin}/login?error=auth_code_error`);
}

 * - Logging automatique (Start/Success/Error)
 */
export function createAction<TInput, TOutput>(config: {
  name: string;
  schema: z.ZodSchema<TInput>;
  handler: (data: TInput) => Promise<TOutput>;
}) {
  return async (rawData: unknown): Promise<ActionResponse<TOutput>> => {
    const actionId = crypto.randomUUID(); // ID unique pour tracer la requÃªte
    
    try {
      // 1. Validation Zod
      const validation = config.schema.safeParse(rawData);
      
      if (!validation.success) {
        const firstError = validation.error.issues[0];
        logger.warn({ 
          action: config.name, 
          actionId,
          error: firstError.message,
          path: firstError.path 
        }, 'Validation failed');
        
        return { error: firstError.message, success: false };
      }

      // 2. ExÃ©cution du handler
      logger.info({ action: config.name, actionId }, 'Action started');
      
      const result = await config.handler(validation.data);
      
      logger.info({ action: config.name, actionId }, 'Action succeeded');
      return { data: result, success: true };

    } catch (error: unknown) {
      // Gestion d'erreur robuste avec vÃ©rification de type
      const errorMessage = error instanceof Error ? error.message : 'Une erreur inconnue est survenue';
      const errorStack = error instanceof Error ? error.stack : undefined;
      
      logger.error({ 
        action: config.name, 
        actionId,
        error: errorMessage,
        stack: errorStack
      }, 'Action failed critical');
      
      // On retourne le message d'erreur
      return { error: errorMessage, success: false };
    }
  };
}
